from collections import deque

# BFS function
def bfs(start, graph):
    visited = set()
    queue = deque()

    queue.append(start)
    visited.add(start)

    while queue:
        node = queue.popleft()
        print(node, end=' ')

        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)

# A* Algorithm for 8 Puzzle using User Input

from heapq import heappush, heappop

# Goal State
GOAL = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 0]
]

# Heuristic Function (Manhattan Distance)
def heuristic(state):
    distance = 0

    for i in range(3):
        for j in range(3):

            value = state[i][j]

            if value != 0:
                goal_x = (value - 1) // 3
                goal_y = (value - 1) % 3

                distance += abs(goal_x - i) + abs(goal_y - j)

    return distance

# Find blank tile position
def find_blank(state):

    for i in range(3):
        for j in range(3):

            if state[i][j] == 0:
                return i, j

# Generate neighbors
def get_neighbors(state):

    neighbors = []

    x, y = find_blank(state)

    directions = [(-1,0), (1,0), (0,-1), (0,1)]

    for dx, dy in directions:

        nx, ny = x + dx, y + dy

        if 0 <= nx < 3 and 0 <= ny < 3:

            new_state = [row[:] for row in state]

            # Swap
            new_state[x][y], new_state[nx][ny] = (
                new_state[nx][ny],
                new_state[x][y]
            )

            neighbors.append(new_state)

    return neighbors

# Convert to tuple
def state_to_tuple(state):
    return tuple(tuple(row) for row in state)

# A* Algorithm
def astar(start):

    pq = []

    heappush(pq, (heuristic(start), 0, start, []))

    visited = set()

    while pq:

        f, g, current, path = heappop(pq)

        if current == GOAL:
            return path + [current]

        current_tuple = state_to_tuple(current)

        if current_tuple in visited:
            continue

        visited.add(current_tuple)

        for neighbor in get_neighbors(current):

            neighbor_tuple = state_to_tuple(neighbor)

            if neighbor_tuple not in visited:

                new_g = g + 1
                new_f = new_g + heuristic(neighbor)

                heappush(
                    pq,
                    (new_f, new_g, neighbor, path + [current])
                )

    return None

# Single Source Shortest Path using Dijkstra Algorithm
# Displays shortest path and distance

import heapq

# Number of vertices
n = int(input("Enter number of vertices: "))

# Number of edges
e = int(input("Enter number of edges: "))

# Graph
graph = {}

for i in range(n):
    graph[i] = []

# Input edges
print("\nEnter edges in format: source destination weight")

for i in range(e):

    u, v, w = map(int, input().split())

    graph[u].append((v, w))
    graph[v].append((u, w))   # Remove for directed graph

# Source vertex
source = int(input("\nEnter source vertex: "))

# Dijkstra Algorithm
distance = [float('inf')] * n
parent = [-1] * n

distance[source] = 0

priority_queue = []

heapq.heappush(priority_queue, (0, source))

while priority_queue:

    current_distance, current_vertex = heapq.heappop(priority_queue)

    for neighbor, weight in graph[current_vertex]:

        new_distance = current_distance + weight

        if new_distance < distance[neighbor]:

            distance[neighbor] = new_distance

            parent[neighbor] = current_vertex

            heapq.heappush(
                priority_queue,
                (new_distance, neighbor)
            )

# Function to print path
def get_path(vertex):

    path = []

    while vertex != -1:
        path.append(vertex)
        vertex = parent[vertex]

    path.reverse()

    return path

# Output

for vertex in range(n):

    print("\nVertex:", vertex)

    print("Shortest Distance:", distance[vertex])

    print("Shortest Path:", " -> ".join(map(str, get_path(vertex))))

# Kruskal's Minimum Spanning Tree Algorithm

# Find Parent
def find(parent, i):
    if parent[i] == i:
        return i
    return find(parent, parent[i])

# Union of sets
def union(parent, rank, x, y):

    xroot = find(parent, x)
    yroot = find(parent, y)

    if rank[xroot] < rank[yroot]:
        parent[xroot] = yroot

    elif rank[xroot] > rank[yroot]:
        parent[yroot] = xroot

    else:
        parent[yroot] = xroot
        rank[xroot] += 1


# User Input
vertices = int(input("Enter number of vertices: "))
edges = int(input("Enter number of edges: "))

graph = []

print("Enter edges in format: source destination weight")

for i in range(edges):
    u, v, w = map(int, input().split())
    graph.append([u, v, w])

# Sort edges by weight
graph = sorted(graph, key=lambda item: item[2])

parent = []
rank = []

for node in range(vertices):
    parent.append(node)
    rank.append(0)

result = []
e = 0
i = 0
total_cost = 0

# Kruskal Algorithm
while e < vertices - 1:

    u, v, w = graph[i]
    i += 1

    x = find(parent, u)
    y = find(parent, v)

    if x != y:
        e += 1
        result.append([u, v, w])
        total_cost += w
        union(parent, rank, x, y)

# Output
print("\nEdges in Minimum Spanning Tree:\n")

for u, v, weight in result:
    print(u, "-", v, "=", weight)

print("\nTotal Cost =", total_cost)

# Input board size
n = int(input("Enter value of N: "))

# Create chess board
board = [["." for _ in range(n)] for _ in range(n)]

# Print board
def print_board():

    print("\n".join([" ".join(row) for row in board]))
    print("-" * 10)


# Check safe position
def is_safe(row, col):

    # Check column
    for i in range(row):

        if board[i][col] == "Q":
            return False

    # Check left diagonal
    i, j = row - 1, col - 1

    while i >= 0 and j >= 0:

        if board[i][j] == "Q":
            return False

        i -= 1
        j -= 1

    # Check right diagonal
    i, j = row - 1, col + 1

    while i >= 0 and j < n:

        if board[i][j] == "Q":
            return False

        i -= 1
        j += 1

    return True


# Backtracking function
def solve(row):

    # Base case
    if row == n:

        print("Solution Found:\n")

        print_board()

        return True

    # Try all columns
    for col in range(n):

        print(f"Trying Row {row}, Col {col}")

        if is_safe(row, col):

            # Place queen
            board[row][col] = "Q"

            print(f"Placed Q at ({row},{col})")

            print_board()

            # Recursive call
            if solve(row + 1):
                return True

            # Backtrack
            board[row][col] = "."

            print(f"Backtracking from ({row},{col})")

            print_board()

    return False


# Start solving
solve(0)

            return False

# Backtracking function
def color_graph(nodes, index):

    # If all nodes are colored
    if index == len(nodes):
        return True

    # Current node
    node = nodes[index]

    # Try all colors
    for color in colors:

        # Check if safe
        if safe(node, color):

            # Assign color
            coloring[node] = color

            # Recursive call
            if color_graph(nodes, index + 1):
                return True

            # Backtracking
            del coloring[node]

    return False


# List of nodes
nodes = list(graph.keys())

# Start graph coloring
if color_graph(nodes, 0):

    print("\nGraph Coloring Solution:\n")

    for node in coloring:
        print(node, "->", coloring[node])

else:
    print("\nNo solution found")

What is EC2?

Amazon EC2 is a web service that provides scalable computing capacity in the cloud. It allows users to create virtual machines quickly without buying physical hardware.

Scalability
          Resources can be increased or decreased as needed.
Elasticity
         Automatic adjustment of resources based on workload.
Security
      Security groups and firewalls protect instances.

           Users pay only for used resources.

           Services are available through multiple regions.
