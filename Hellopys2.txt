import numpy as np

def distance (lat1, lon1, lat2, lon2):
    lon1,lon2,lat1,lat2 = map(np.radians, [lon1,lon2,lat1,lat2])

    diff_lon = lon2 - lon1
    diff_lat = lat2 - lat1

    km = 2*6371 * np.arcsin(np.sqrt(np.sin(diff_lat/2.0)**2 + np.cos(lat1)* np.cos(lat2)* np.sin(diff_lon/2.0)**2))
    return km

import numpy as np
import matplotlib.pyplot as plt

def f(x):
    return x**2 + 6*x+9


def df(x):
    return 2*x+6 


def gradient_descent(starting_point, learning_rate, iterations): 
    x = starting_point
    for i in range(iterations):
        x = x - learning_rate * df(x)  # update step
        print(f"Iteration {i+1}: x = {x:.4f}, f(x) = {f(x):.4f}") 
    return x 
starting_point = 2 
learning_rate = 0.01 
iterations = 100 
minimum = gradient_descent(starting_point, learning_rate, iterations)
print(f"\nLocal minimum occurs at x = {minimum:.4f}, f(x) = {f(minimum):.4f}")




x_vals = np.linspace(-10, 2, 100) 
y_vals = f(x_vals)

import numpy as np 
from sklearn.base import BaseEstimator, TransformerMixin



class IQRCapper(BaseEstimator, TransformerMixin):
    def fit(self, X, y=None):        
      X = np.asarray(X, dtype=float)        
      q1 = np.nanpercentile(X, 25, axis=0)        
      q3 = np.nanpercentile(X, 75, axis=0)        
      iqr = q3 - q1

      self.lower_ = q1 - 1.5 * iqr        
      self.upper_ = q3 + 1.5 * iqr

      return self

    def transform(self, X):        
      X = np.asarray(X, dtype=float)        
      return np.clip(X, self.lower_, self.upper_)
      
("IQRCapper class defined successfully!")

outlier_bounds = {}
for col in features:
    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)
    IQR = Q3 - Q1
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR
    outlier_bounds[col] = (lower_bound, upper_bound)
    print(
        f"{col}: Q1={Q1:.2f}, Q3={Q3:.2f}, "
        f"IQR={IQR:.2f}, "
        f"Lower={lower_bound:.2f}, "
        f"Upper={upper_bound:.2f}"
    )

    outlier_mask = (
        (df[col] < lower) |
        (df[col] > upper)
    )
    outlier_counts[col] = int(outlier_mask.sum())
print("\nOutlier count by feature:")
